async function fetchRuIP() { const el = document.getElementById('ru-ip'); const apiKey = 'ent_dd02bd27037e0d9ab94b7019665b9ff457251b9290227179'; // 1. Запрос к открытому роуту Enterno (возвращает IP клинета) try { const res = await fetch('https://enterno.io/api/open/ip'); if (res.ok) { const json = await res.json(); const ip = json.ip || (json.data && json.data.ip) || json.query; if (ip) { el.textContent = ip; return; } } } catch (e) {} // 2. Запрос к API v4 try { const res = await fetch(`https://enterno.io/api/v4/ip?key=${apiKey}`); if (res.ok) { const json = await res.json(); const ip = (json.data && json.data.ip) || json.ip || json.query; if (ip) { el.textContent = ip; return; } } } catch (e) {} el.textContent = 'offline'; }
